<?php

$http = new OpenSwoole\Http\Server('127.0.0.1', 9502);

$jsonResponse = json_encode([
    'message' => 'Hello World',
]);

$http->on('request', function ($request, $response) use ($jsonResponse) {
    if (
        $request->server['request_uri'] === '/hello' &&
        $request->server['request_method'] === 'GET'
    ) {
        $response->header('Content-Type', 'application/json');
        $response->status(200);
        $response->end($jsonResponse);

        return;
    }

    $response->status(404);
    $response->header('Content-Type', 'application/json');
    $response->end(json_encode([
        'message' => 'Not Found',
    ]));
});

echo "OpenSwoole listening on http://127.0.0.1:9502\n";

$http->start();
